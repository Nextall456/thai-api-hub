package main

import (
    "context"
    "fmt"
    openai "github.com/sashabaranov/go-openai"
)

func main() {
    cfg := openai.DefaultConfig("th-xxxxxxxxxxxxxxxx")
    cfg.BaseURL = "{{ site_url }}/v1"
    client := openai.NewClientWithConfig(cfg)

    resp, _ := client.CreateChatCompletion(
        context.Background(),
        openai.ChatCompletionRequest{
            Model: "thai-hub/auto",
            Messages: []openai.ChatCompletionMessage{
                {Role: openai.ChatMessageRoleUser, Content: "Hello from Go!"},
            },
        },
    )
    fmt.Println(resp.Choices[0].Message.Content)
}

คำถามที่พบบ่อย (Developer FAQ)

เปลี่ยนมาใช้ Thai API Hub ต้องแก้โค้ดเยอะไหม?

ไม่เลยครับ โค้ดเดิมที่ใช้ OpenAI SDK (Python, Node.js, Go, PHP) แก้เพียง 2 บรรทัด คือ base_url="https://thai-api-hub-production.up.railway.app/v1" และ api_key="th-คีย์ของคุณ" ก็สามารถใช้งาน Streaming, Tool Calling และ Responses ได้ทันที

ระบบสลับโมเดลอัตโนมัติ (Circuit Breaker Failover) ทำงานอย่างไร?

เรามี In-Memory Circuit Breaker คอยวัดผลความเร็วและสถานะของแต่ละ Backend หากโมเดลตัวแรกล้มเหลวหรือติด Rate Limit ระบบจะตัดวงจรและ Failover สลับไปยังโมเดลสำรองที่มีคุณภาพใกล้เคียงกันทันทีในเวลา <5ms โดยไม่ต้องรอ HTTP Timeout นาน 15-30 วินาที

มี Response Headers อะไรส่งกลับมาบ้างสำหรับ Observability?

ทุก Request จะมี Header มาตรฐานส่งกลับมาเสมอ ได้แก่ X-Thai-Hub-Backend (บอก provider และ model จริงที่ประมวลผล), X-Thai-Hub-Latency-Ms, X-Thai-Hub-Attempts, และกลุ่ม X-RateLimit-* แสดงจำนวนคำขอและ Token คงเหลือ

การชำระเงินและการต่ออายุทำอย่างไร?

ชำระผ่าน PromptPay QR หรือโอนผ่านธนาคารไทยได้ทุกธนาคารในหน้า Billing ในแดชบอร์ด ไม่มีค่าธรรมเนียมบัตรเครดิตต่างประเทศ ไม่มีระบบหักเงินอัตโนมัติ หมดอายุแล้วระบบจะหยุดทำงานโดยไม่มีบิลค้าง
